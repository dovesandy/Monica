1. 算术运算符

---------

### 基本算术运算符

``` cpp
#include <iostream>
using namespace std;

int main() {
    int a = 10, b = 3;
    
    cout << "a = " << a << ", b = " << b << endl;
    cout << "加法: a + b = " << a + b << endl;        // 13
    cout << "减法: a - b = " << a - b << endl;        // 7
    cout << "乘法: a * b = " << a * b << endl;        // 30
    cout << "除法: a / b = " << a / b << endl;        // 3 (整数除法)
    cout << "取模: a % b = " << a % b << endl;        // 1
    
    // 浮点数除法
    double x = 10.0, y = 3.0;
    cout << "浮点除法: x / y = " << x / y << endl;    // 3.33333
    
    return 0;
}
```

### 整数除法的注意事项

``` cpp
#include <iostream>
using namespace std;

int main() {
    int result1 = 7 / 2;      // 结果为3，不是3.5
    int result2 = 9 / 4;      // 结果为2，不是2.25
    
    cout << "7 / 2 = " << result1 << endl;
    cout << "9 / 4 = " << result2 << endl;
    
    // 如果要得到小数结果，需要至少一个操作数是浮点数
    double result3 = 7.0 / 2; // 结果为3.5
    double result4 = 9 / 4.0; // 结果为2.25
    
    cout << "7.0 / 2 = " << result3 << endl;
    cout << "9 / 4.0 = " << result4 << endl;
    
    return 0;
}
```

2. 关系运算符

---------

### 比较运算

``` cpp
#include <iostream>
using namespace std;

int main() {
    int a = 10, b = 5, c = 10;
    
    cout << "a = " << a << ", b = " << b << ", c = " << c << endl;
    cout << "相等比较: a == c = " << (a == c) << endl;    // 1 (true)
    cout << "不等比较: a != b = " << (a != b) << endl;    // 1 (true)
    cout << "大于比较: a > b = " << (a > b) << endl;      // 1 (true)
    cout << "小于比较: a < b = " << (a < b) << endl;      // 0 (false)
    cout << "大于等于: a >= c = " << (a >= c) << endl;    // 1 (true)
    cout << "小于等于: b <= a = " << (b <= a) << endl;    // 1 (true)
    
    return 0;
}
```

3. 逻辑运算符

---------

### 与(&&)、或(||)、非(!)运算

``` cpp
#include <iostream>
using namespace std;

int main() {
    bool isSunny = true;
    bool isWeekend = false;
    int temperature = 25;
    
    cout << "天气晴朗: " << isSunny << endl;
    cout << "是周末: " << isWeekend << endl;
    cout << "温度: " << temperature << "°C" << endl;
    
    // 逻辑与：两个条件都为真时结果为真
    bool goodWeather = isSunny && (temperature > 20);
    cout << "好天气(晴朗且温度>20): " << goodWeather << endl;
    
    // 逻辑或：至少一个条件为真时结果为真
    bool canGoOut = isWeekend || goodWeather;
    cout << "可以外出(周末或好天气): " << canGoOut << endl;
    
    // 逻辑非：取反
    bool isRainy = !isSunny;
    cout << "下雨天: " << isRainy << endl;
    
    // 复杂逻辑表达式
    bool perfectDay = isSunny && (temperature > 22) && (temperature < 30);
    cout << "完美天气(晴朗且22-30度): " << perfectDay << endl;
    
    return 0;
}
```

4. 赋值运算符

---------

### 基本赋值和复合赋值

``` cpp
#include <iostream>
using namespace std;

int main() {
    int x = 10;
    cout << "初始值 x = " << x << endl;
    
    // 基本赋值
    x = 20;
    cout << "赋值后 x = " << x << endl;
    
    // 复合赋值运算符
    x += 5;  // 等价于 x = x + 5
    cout << "x += 5 后: " << x << endl;
    
    x -= 3;  // 等价于 x = x - 3
    cout << "x -= 3 后: " << x << endl;
    
    x *= 2;  // 等价于 x = x * 2
    cout << "x *= 2 后: " << x << endl;
    
    x /= 4;  // 等价于 x = x / 4
    cout << "x /= 4 后: " << x << endl;
    
    x %= 3;  // 等价于 x = x % 3
    cout << "x %= 3 后: " << x << endl;
    
    return 0;
}
```

5. 自增和自减运算符

------------

### 前缀和后置的区别

``` cpp
#include <iostream>
using namespace std;

int main() {
    int a = 5;
    
    // 后缀自增：先使用值，再自增
    cout << "a++ 的结果: " << a++ << endl;  // 输出5，然后a变为6
    cout << "现在 a 的值: " << a << endl;   // 输出6
    
    // 前缀自增：先自增，再使用值
    cout << "++a 的结果: " << ++a << endl;  // a先变为7，然后输出7
    
    int b = 10;
    
    // 后缀自减
    cout << "b-- 的结果: " << b-- << endl;  // 输出10，然后b变为9
    cout << "现在 b 的值: " << b << endl;   // 输出9
    
    // 前缀自减
    cout << "--b 的结果: " << --b << endl;  // b先变为8，然后输出8
    
    // 在实际使用中的例子
    int counter = 0;
    cout << "计数: " << counter++ << endl;  // 输出0，然后counter=1
    cout << "计数: " << counter++ << endl;  // 输出1，然后counter=2
    cout << "计数: " << counter++ << endl;  // 输出2，然后counter=3
    
    return 0;
}
```

6. 运算符优先级

----------

### 优先级示例

``` cpp
#include <iostream>
using namespace std;

int main() {
    int a = 10, b = 5, c = 2;
    
    // 不同优先级的运算
    int result1 = a + b * c;      // 先乘法，后加法：10 + (5*2) = 20
    int result2 = (a + b) * c;    // 先括号，后乘法：(10+5)*2 = 30
    int result3 = a * b / c;      // 从左到右：(10*5)/2 = 25
    int result4 = a + b > c;      // 先算术，后比较：(10+5)>2 = 1(true)
    
    cout << "a + b * c = " << result1 << endl;
    cout << "(a + b) * c = " << result2 << endl;
    cout << "a * b / c = " << result3 << endl;
    cout << "a + b > c = " << result4 << endl;
    
    // 逻辑运算符优先级
    bool logicResult = a > b && b > c || a < c;
    // 等价于：(a > b) && (b > c) || (a < c)
    
    cout << "a > b && b > c || a < c = " << logicResult << endl;
    
    return 0;
}
```

7. 类型转换

--------

### 隐式类型转换

``` cpp
#include <iostream>
using namespace std;

int main() {
    // 整数提升
    short s = 10;
    int i = s;  // short自动转换为int
    
    // 浮点数转换
    int integer = 10;
    double decimal = integer;  // int自动转换为double
    
    cout << "short转int: " << i << endl;
    cout << "int转double: " << decimal << endl;
    
    // 混合类型运算
    int a = 10;
    double b = 3.5;
    double result = a + b;  // a自动转换为double
    
    cout << "10 + 3.5 = " << result << endl;
    
    return 0;
}
```

### 显式类型转换（强制转换）

``` cpp
#include <iostream>
using namespace std;

int main() {
    double pi = 3.14159;
    
    // C风格强制转换
    int intPi = (int)pi;
    cout << "pi的整数部分: " << intPi << endl;
    
    // C++风格强制转换
    int intPi2 = static_cast<int>(pi);
    cout << "使用static_cast: " << intPi2 << endl;
    
    // 整数转浮点数
    int number = 100;
    double doubleNumber = static_cast<double>(number);
    cout << "整数转浮点数: " << doubleNumber << endl;
    
    return 0;
}
```

8. 实践练习

--------

### 练习1：计算器程序

``` cpp
#include <iostream>
using namespace std;

int main() {
    double num1, num2;
    char operation;
    
    cout << "=== 增强版计算器 ===" << endl;
    cout << "请输入第一个数字: ";
    cin >> num1;
    
    cout << "请输入运算符 (+, -, *, /): ";
    cin >> operation;
    
    cout << "请输入第二个数字: ";
    cin >> num2;
    
    double result;
    bool validOperation = true;
    
    switch(operation) {
        case '+':
            result = num1 + num2;
            break;
        case '-':
            result = num1 - num2;
            break;
        case '*':
            result = num1 * num2;
            break;
        case '/':
            if (num2 != 0) {
                result = num1 / num2;
            } else {
                cout << "错误：除数不能为0！" << endl;
                validOperation = false;
            }
            break;
        default:
            cout << "错误：不支持的运算符！" << endl;
            validOperation = false;
    }
    
    if (validOperation) {
        cout << num1 << " " << operation << " " << num2 << " = " << result << endl;
    }
    
    return 0;
}
```

### 练习2：成绩判断程序

``` cpp
#include <iostream>

int main() {
```

9. 今日总结

--------

### 重点回顾

* ✅ **算术运算符**: `+`, `-`, `*`, `/`, `%`

* ✅ **关系运算符**: `==`, `!=`, `>`, `<`, `>=`, `<=`

* ✅ **逻辑运算符**: `&&`, `||`, `!`

* ✅ **赋值运算符**: `=`, `+=`, `-=`, `*=`, `/=`, `%=`

* ✅ **自增自减**: `++`, `--`（前缀和后置的区别）

* ✅ **运算符优先级**: 括号 > 算术 > 关系 > 逻辑 > 赋值

* ✅ **类型转换**: 隐式转换和显式转换

### 常见错误提醒

1. **整数除法**: `5 / 2` 结果是 `2`，不是 `2.5`

2. **取模运算**: 只能用于整数，`5.5 % 2` 是错误的

3. **逻辑运算短路**: `&&` 和 `||` 有短路求值特性

4. **自增顺序**: `a++` 和 `++a` 有重要区别
